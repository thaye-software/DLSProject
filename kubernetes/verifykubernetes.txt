# Paste this whole file into PowerShell. Each line prints OK or FAIL. The last line is ALL OK only when every check passed.
$fail = 0
if ((docker info --format '{{.ServerVersion}}' 2>$null)) { 'OK  Docker Desktop engine' } else { $fail++; 'FAIL  Docker Desktop engine' }
if ((kubectl config current-context 2>$null) -eq 'docker-desktop') { 'OK  kubectl context docker-desktop' } else { $fail++; 'FAIL  kubectl context docker-desktop' }
if ((kubectl get node docker-desktop -o jsonpath="{.status.conditions[?(@.type=='Ready')].status}" 2>$null) -eq 'True') { 'OK  node docker-desktop Ready' } else { $fail++; 'FAIL  node docker-desktop Ready' }
foreach ($d in 'rabbitmq','catalog-db','catalog','orders-db','orders','payment','notif','search-db','search-cache','search','messaging-db','messaging') { if ((kubectl get deploy $d -n default -o jsonpath='{.status.readyReplicas}' 2>$null) -eq '1') { "OK  deployment $d running" } else { $fail++; "FAIL  deployment $d running" } }
if ((kubectl get deploy rabbitmq -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null) -eq 'rabbitmq:4-management IfNotPresent') { 'OK  rabbitmq image' } else { $fail++; 'FAIL  rabbitmq image' }
if ((kubectl get deploy catalog-db -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null) -eq 'postgres:17-alpine IfNotPresent') { 'OK  catalog-db image' } else { $fail++; 'FAIL  catalog-db image' }
if ((kubectl get deploy orders-db -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null) -eq 'postgres:17-alpine IfNotPresent') { 'OK  orders-db image' } else { $fail++; 'FAIL  orders-db image' }
if ((kubectl get deploy search-db -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null) -eq 'mongo:7 IfNotPresent') { 'OK  search-db image' } else { $fail++; 'FAIL  search-db image' }
if ((kubectl get deploy messaging-db -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null) -eq 'mongo:7 IfNotPresent') { 'OK  messaging-db image' } else { $fail++; 'FAIL  messaging-db image' }
if ((kubectl get deploy search-cache -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null) -eq 'redis:7-alpine IfNotPresent') { 'OK  search-cache image' } else { $fail++; 'FAIL  search-cache image' }
foreach ($img in 'catalog=dls/catalog:local','orders=dls/orders:local','payment=dls/payment:local','notif=dls/notif:local','search=dls/search:local','messaging=dls/messaging:local') { $n,$want = $img.Split('='); $got = kubectl get deploy $n -n default -o jsonpath='{.spec.template.spec.containers[0].image} {.spec.template.spec.containers[0].imagePullPolicy}' 2>$null; if ($got -eq "$want Never") { "OK  $n image" } else { $fail++; "FAIL  $n image" } }
if ((kubectl get configmap rabbitmq-config -n default -o jsonpath='{.data.rabbitmq\.conf}' 2>$null) -match 'loopback_users.guest = false') { 'OK  configmap rabbitmq-config' } else { $fail++; 'FAIL  configmap rabbitmq-config' }
foreach ($k in 'catalog-secret=POSTGRES_USER,POSTGRES_PASSWORD,POSTGRES_DB,DATABASE_URL,INTERNAL_JWT_SECRET','orders-secret=POSTGRES_USER,POSTGRES_PASSWORD,POSTGRES_DB,DATABASE_URL,INTERNAL_JWT_SECRET','payment-secret=INTERNAL_JWT_SECRET') { $n,$keys = $k.Split('=',2); $missing = @($keys.Split(',') | Where-Object { -not (kubectl get secret $n -n default -o jsonpath="{.data.$_}" 2>$null) }); if ($missing.Count -eq 0) { "OK  secret $n" } else { $fail++; "FAIL  secret $n" } }
foreach ($p in 'catalog-db-data','orders-db-data','search-db-data','messaging-db-data') { if ((kubectl get pvc $p -n default -o jsonpath='{.status.phase}' 2>$null) -eq 'Bound') { "OK  pvc $p Bound" } else { $fail++; "FAIL  pvc $p Bound" } }
foreach ($n in 'rabbitmq','catalog','catalog-db','orders','orders-db','payment','notif','search','search-db','search-cache','messaging','messaging-db') { if ((kubectl get svc $n -n default -o jsonpath='{.spec.type}' 2>$null) -eq 'ClusterIP') { "OK  service $n ClusterIP" } else { $fail++; "FAIL  service $n ClusterIP" } }
if ((kubectl get svc gateway -n default -o jsonpath='{.spec.type} {.spec.ports[0].nodePort}' 2>$null) -eq 'NodePort 30080') { 'OK  service gateway nodePort 30080' } else { $fail++; 'FAIL  service gateway nodePort 30080' }
if ((curl.exe -s --max-time 5 'http://localhost:30080/health') -match '"status"\s*:\s*"ok"') { 'OK  gateway health http://localhost:30080/health' } else { $fail++; 'FAIL  gateway health http://localhost:30080/health' }
if ($fail -eq 0) { 'ALL OK' } else { "NOT ALL OK  ($fail failed)" }
